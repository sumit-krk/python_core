# Loop Examples in Python

## Practical Loop Examples

### Example 1: Multiplication Table
```python
# Print 5x5 multiplication table
for i in range(1, 6):
    for j in range(1, 6):
        print(f"{i*j:3}", end=" ")
    print()
```

### Example 2: Fibonacci Sequence
```python
# Generate first N Fibonacci numbers
n = 10
a, b = 0, 1

for _ in range(n):
    print(a, end=" ")
    a, b = b, a + b
```

### Example 3: Sum of Numbers
```python
# Sum numbers from 1 to 100
total = 0
for i in range(1, 101):
    total += i
print(f"Sum: {total}")
```

### Example 4: List Processing
```python
# Process list of numbers
numbers = [1, 2, 3, 4, 5]
squared = []

for num in numbers:
    squared.append(num ** 2)

print(squared)  # [1, 4, 9, 16, 25]
```

### Example 5: Find Maximum
```python
# Find maximum in list
numbers = [3, 7, 2, 9, 1, 5]
max_num = numbers[0]

for num in numbers:
    if num > max_num:
        max_num = num

print(f"Maximum: {max_num}")
```

### Example 6: Count Occurrences
```python
# Count occurrences of each character
text = "hello world"
char_count = {}

for char in text:
    if char in char_count:
        char_count[char] += 1
    else:
        char_count[char] = 1

print(char_count)
```

### Example 7: Reverse String
```python
# Reverse a string
text = "Python"
reversed_text = ""

for char in text:
    reversed_text = char + reversed_text

print(reversed_text)  # nohtyP
```

### Example 8: Check Palindrome
```python
# Check if string is palindrome
text = "racecar"
is_palindrome = True

for i in range(len(text) // 2):
    if text[i] != text[len(text) - 1 - i]:
        is_palindrome = False
        break

print(f"Palindrome: {is_palindrome}")
```

### Example 9: Prime Numbers
```python
# Find prime numbers up to N
n = 20

for num in range(2, n + 1):
    is_prime = True
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            is_prime = False
            break
    if is_prime:
        print(num, end=" ")
```

### Example 10: Pattern Printing
```python
# Print pyramid pattern
n = 5
for i in range(1, n + 1):
    for j in range(1, i + 1):
        print("*", end=" ")
    print()
```
